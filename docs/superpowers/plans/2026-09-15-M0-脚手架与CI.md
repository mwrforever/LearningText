# M0 工程脚手架 + CI 门禁落地 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.
>
> **执行模式（用户已指定）：SDD（superpowers:subagent-driven-development）**——由主控会话派发子代理逐任务执行，每任务一个全新子代理 + 两阶段评审。所有工作在 **dev 分支**进行（`git checkout dev`）；**main 是生产分支，项目完整落地后才合入并触发完整 CI/CD 打包发布**。

**Goal:** 仓库具备可运行的 Electron + TypeScript 最小应用（窗口 + 一条类型化 IPC 通道），lint/format/typecheck/test 全绿，GitHub Actions 三平台矩阵门禁通过，dev/main 分支保护生效。

**Architecture:** 主进程（tsc 编译，CJS）+ 渲染进程（Vite 构建 React，ESM）+ preload 桥 + shared 契约；生产环境经 `app://` 自定义协议加载本地产物（禁 file://），开发环境经 Vite dev server；`system:ping` 通道作为 A.7 类型化 IPC 模式的样板（zod 入口校验 + senderFrame origin 白名单 + Result DTO 返回）。

**Tech Stack:** Electron 44.3.x · TypeScript 7.0.x（strict）· React 19.3.x + Vite 8.3.x · better-sqlite3 13.0.x（WAL 冒烟）· zod 4.6.x · Vitest 5.0.x + @vitest/coverage-v8（严格同版）· Playwright 1.63.x（`_electron`）· electron-builder 26.15.x · ESLint 10 flat config + Prettier · husky 9 + lint-staged 17 · Node 24 LTS。

## Global Constraints（摘自宪法 AGENTS.md，每任务隐含遵守）

- 全部代码/注释/日志**中文**；UTF-8 无 BOM、**LF 行尾**（.gitattributes 已强制）；禁止 CRLF。
- TS `strict: true` + `noUncheckedIndexedAccess: true`；禁隐式/显式 `any`（`unknown` + 收窄）；禁在 zod/IPC 边界用 `!`/`as`。
- 版本**精确锁定**（无 `^`/`~`），以 `AGENTS.md` C.2 基线为准；最终在 package-lock.json 固化。
- 渲染窗口安全基线（B.5）：`contextIsolation: true`、`sandbox: true`、`nodeIntegration: false` 显式写出；禁 `loadURL` 外部 URL；渲染页 CSP `default-src 'self'`；每通道 `senderFrame` origin 白名单 + zod `safeParse`；错误经 Result DTO。
- 目录以 `AGENTS.md` B.1 注释式树为准；产物输出 `dist/`（main/preload/renderer）与 `release/`，均 gitignore。
- npm scripts 必须与 `AGENTS.md` C.4 代码块对齐（新增 `start:dev`、`dev:renderer`、`dev:main` 属 C.4 对齐核对回填项，落地后在 TASK.md 记录）。
- CI：方案 A 三平台矩阵（ubuntu/windows/macos），六阶段，触发面 `[main, dev]`；`HUSKY=0` 安装；失败即红，禁止绕过。
- 覆盖率门禁（A.6-3）：`coverage.include: ['src/**/*.{ts,tsx}']`；核心（src/main、src/shared、src/preload）100%，全局 ≥80%；`@vitest/coverage-v8` 与 vitest 版本严格一致。

```ts
/**
 * Result 型 DTO（宪法 A.7-3）：跨进程错误显式建模，禁依赖异常透传。
 * AppError.code 取值见 src/shared/errors.ts；message 面向用户、必须中文。
 */
export interface AppError {
  /** 业务错误码，如 E_IPC_BAD_PAYLOAD */
  code: string;
  /** 面向用户的中文消息 */
  message: string;
}

/** 请求结果的统一载体：成功带 value，失败带 error，两分支互斥 */
export type Result<T> = { ok: true; value: T } | { ok: false; error: AppError };

/** 构造成功结果 */
export const ok = <T>(value: T): Result<T> => ({ ok: true, value });

/** 构造失败结果：code 用 errors.ts 中的常量，禁裸写字符串 */
export const err = (code: string, message: string): Result<never> => ({
  ok: false,
  error: { code, message },
});
```

```ts
// app:// 静态资源路径解析（纯函数，禁止 import electron，保证可单测）
import path from 'node:path';

/** 扩展名 → Content-Type 映射；未命中返回 octet-stream 由浏览器按下载处理 */
const CONTENT_TYPES: Record<string, string> = {
  '.html': 'text/html; charset=utf-8',
  '.js': 'text/javascript; charset=utf-8',
  '.css': 'text/css; charset=utf-8',
  '.json': 'application/json; charset=utf-8',
  '.svg': 'image/svg+xml',
  '.png': 'image/png',
  '.jpg': 'image/jpeg',
  '.ico': 'image/x-icon',
  '.woff2': 'font/woff2',
  '.map': 'application/json; charset=utf-8',
};

export interface AppResource {
  /** 文件绝对路径 */
  abs: string;
  /** 响应 Content-Type */
  contentType: string;
}

/**
 * 将 app:// 的 URL pathname 解析为 dist 内的文件路径。
 * 返回 null 表示拒绝（路径穿越 / 不存在），由协议层转 404。
 * @param pathname 如 "/" 或 "/assets/a.js"（已去掉 query/hash）
 * @param distRoot 渲染产物根目录（绝对路径）
 */
export function resolveAppPath(pathname: string, distRoot: string): AppResource | null {
  // URL 解码后校验：防 %2e%2e 形式的穿越
  let decoded: string;
  try {
    decoded = decodeURIComponent(pathname);
  } catch {
    return null;
  }
  const relative = decoded === '/' ? 'index.html' : decoded.replace(/^\//, '');
  const abs = path.resolve(distRoot, relative);
  // 解析结果必须仍落在 distRoot 内（B.5-2：禁逃逸 VFS/产物目录）
  if (abs !== distRoot && !abs.startsWith(distRoot + path.sep)) {
    return null;
  }
  const contentType = CONTENT_TYPES[path.extname(abs).toLowerCase()];
  if (contentType === undefined) {
    return null;
  }
  return { abs, contentType };
}
```

> 注：`noUncheckedIndexedAccess` 下 `CONTENT_TYPES[...]` 读取为 `string | undefined`，用显式 undefined 判断收窄，不用 `!`。

```ts
import path from 'node:path';
import { app, net } from 'electron';
import { resolveAppPath } from './appPaths';

/** 渲染产物根目录：<安装目录>/dist/renderer */
const distRoot = path.join(app.getAppPath(), 'dist', 'renderer');

/** protocol.handle 处理器：资源缺失或越界一律 404，不泄露文件系统信息 */
export function handleAppResource(request: Request): Promise<Response> {
  const url = new URL(request.url);
  const resource = resolveAppPath(url.pathname, distRoot);
  if (resource === null) {
    return Promise.resolve(new Response('Not Found', { status: 404 }));
  }
  return net.fetch('file://' + resource.abs, {
    headers: { 'Content-Type': resource.contentType },
  });
}
```

```ts
/**
 * IPC handler 集中注册（宪法 B.3-2）：
 * 每通道两道校验——senderFrame origin 白名单（B.5-6）→ zod safeParse（A.7-5），
 * 校验失败与业务结果统一以 Result DTO 返回（A.7-3），禁止抛异常透传。
 */
import { ipcMain } from 'electron';
import { z } from 'zod';
import { IPC } from '../shared/ipc';
import { err, ok, type Result } from '../shared/result';
import { E_IPC_BAD_PAYLOAD, E_IPC_FORBIDDEN_ORIGIN } from '../shared/errors';

/** system:ping 请求载荷：探针通道固定为 null，其余一律拒绝 */
const PingPayloadSchema = z.null();

export interface IpcHandlerDeps {
  /** 允许发起 IPC 的 origin 白名单（用 origin 不用 URL，B.5-6） */
  allowedOrigins: readonly string[];
}

export function registerIpcHandlers(deps: IpcHandlerDeps): void {
  ipcMain.handle(IPC.systemPing, (event, payload: unknown): Result<{ pong: true }> => {
    // senderFrame 可能为 null（B.5-6），null 与非白名单 origin 一律拒绝
    const origin = event.senderFrame?.origin;
    if (origin === undefined || origin === null || origin === '' || !deps.allowedOrigins.includes(origin)) {
      return err(E_IPC_FORBIDDEN_ORIGIN, '拒绝来自未授权来源的调用');
    }
    const parsed = PingPayloadSchema.safeParse(payload);
    if (!parsed.success) {
      return err(E_IPC_BAD_PAYLOAD, '请求载荷不合法');
    }
    return ok({ pong: true });
  });
}
```

- [ ] **Step 2: 写入 .github/workflows/ci.yml**

```yaml
# CI 门禁：方案 A 三平台全量严格矩阵（宪法 C.5；细则依据 docs/agmds-research/2026-09-14-CI链与生产落地.md §三）
# 触发面：main 为生产分支（项目完整落地后合入，届时叠加 release 工作流），dev 为日常开发主干
name: CI

on:
  pull_request:
    branches: [main, dev]
  push:
    branches: [main, dev]

# 旧跑立即取消；权限最小化（只读，无发布）
concurrency:
  group: ${{ github.workflow }}-${{ github.ref }}
  cancel-in-progress: true

permissions:
  contents: read

jobs:
  build:
    name: build (${{ matrix.os }})
    runs-on: ${{ matrix.os }}
    timeout-minutes: 45
    strategy:
      fail-fast: false
      matrix:
        os: [ubuntu-latest, windows-latest, macos-latest]
    env:
      # CI 内禁用 husky 钩子（宪法 C.6-2）
      HUSKY: 0
    steps:
      - uses: actions/checkout@v5

      - uses: actions/setup-node@v5
        with:
          node-version-file: .nvmrc
          cache: npm

      # 缓存 Electron 二进制与 electron-builder 下载（key 含 OS，路径不存在时自动跳过）
      - uses: actions/cache@v4
        with:
          key: electron-${{ runner.os }}-${{ hashFiles('package-lock.json') }}
          path: |
            ~/.cache/electron
            ~/.cache/electron-builder
            ~/AppData/Local/electron/Cache
            ~/AppData/Local/electron-builder/Cache
            ~/Library/Caches/electron
            ~/Library/Caches/electron-builder

      - name: 安装依赖
        run: npm ci

      - name: 静态质量门禁（格式 / lint / 类型）
        run: |
          npm run format:check
          npm run lint
          npm run typecheck

      - name: 单元 + 集成测试与覆盖率门禁
        run: npm run test:coverage

      - name: 原生模块 ABI 重编译（最高风险点，三平台全验）
        run: npm run rebuild

      - name: 完整构建（主进程 + 渲染层）
        run: npm run build

      # Linux runner 无显示服务，E2E 需 Xvfb；系统依赖按官方 install-deps 安装
      - name: 安装 E2E 系统依赖（仅 Linux）
        if: runner.os == 'Linux'
        run: npx playwright install-deps

      - name: E2E（_electron 生产形态）
        run: |
          if [ "$RUNNER_OS" = "Linux" ]; then
            xvfb-run --auto-servernum npx playwright test
          else
            npx playwright test
          fi
        shell: bash

      - name: 打包冒烟（--dir 不签名不发布）
        run: npx electron-builder --dir --publish never

      - name: 上传 E2E 报告（失败排查用，不拦截门禁）
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: e2e-report-${{ matrix.os }}
          path: |
            playwright-report/
            test-results/
          retention-days: 7
          if-no-files-found: ignore
```
